Note: Using 'use strict' on top can prevent accidentally leaking values or references to the global space where anyone can access, which can lead to memory leaks. Additionally, it prevents accidental variable declarations that might overwrite global variables. 

        <>
        "use script"

        //global variable
        var a = 10;

        //c is global
        function abc(){
          var  c = "Hi Sagar Kudu" //declare the variable
        }
        abc()
        console.log(window.c)

        //d is still global here
        function abcd(){
            this.d = "I love india"
        }
        new abcd() //create new instance 
        console.log(window.d)
